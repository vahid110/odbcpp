cmake_minimum_required(VERSION 3.20)
project(ODBCPPLegacyInstallConsumer LANGUAGES CXX)
if(NOT CMAKE_GENERATOR STREQUAL "Unix Makefiles")
  message(FATAL_ERROR "Only selected Unix Makefiles command/depfile grammar")
endif()
if(NOT CMAKE_CXX_COMPILER_ID MATCHES "^(GNU|Clang|AppleClang)$")
  message(FATAL_ERROR "Full dependency/argv grammar unselected for this compiler")
endif()
if(NOT IS_ABSOLUTE "${LEGACY_PREFIX}" OR NOT IS_DIRECTORY "${LEGACY_PREFIX}" OR
   NOT LEGACY_INCLUDEDIR OR NOT LEGACY_LIBDIR)
  message(FATAL_ERROR "Explicit installed prefix and GNU directories required")
endif()
if(IS_ABSOLUTE "${LEGACY_INCLUDEDIR}" OR IS_ABSOLUTE "${LEGACY_LIBDIR}" OR
   LEGACY_INCLUDEDIR MATCHES "(^|/)\\.\\.(/|$)" OR LEGACY_LIBDIR MATCHES "(^|/)\\.\\.(/|$)")
  message(FATAL_ERROR "Relative nonescaping GNU directories required")
endif()
foreach(_required IN ITEMS LEGACY_FORBIDDEN_ROOTS LEGACY_SYSTEM_ROOTS LEGACY_TOOLCHAIN_ROOTS)
  if(NOT ${_required})
    message(FATAL_ERROR "Explicit forbidden source/build and system/toolchain roots required")
  endif()
endforeach()
file(REAL_PATH "${LEGACY_PREFIX}" _prefix_identity)
function(_legacy_prefix_identity path output)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Package input missing")
  endif()
  file(REAL_PATH "${path}" _identity)
  file(RELATIVE_PATH _inside "${_prefix_identity}" "${_identity}")
  if(IS_ABSOLUTE "${_inside}" OR _inside MATCHES "^\\.\\.(/|$)")
    message(FATAL_ERROR "Canonical package input escapes prefix")
  endif()
  set(${output} "${_identity}" PARENT_SCOPE)
endfunction()
_legacy_prefix_identity("${LEGACY_PREFIX}/${LEGACY_LIBDIR}/cmake/odbcpp" _package)
foreach(_file IN ITEMS odbcppConfig.cmake odbcppConfigVersion.cmake odbcppTargets.cmake odbcppTargets-release.cmake)
  _legacy_prefix_identity("${_package}/${_file}" _verified_file)
endforeach()
if(DEFINED odbcpp_DIR)
  _legacy_prefix_identity("${odbcpp_DIR}" _override)
  if(NOT _override STREQUAL _package)
    message(FATAL_ERROR "Preselected package override escapes prefix")
  endif()
endif()
find_package(odbcpp CONFIG REQUIRED PATHS
  "${LEGACY_PREFIX}/${LEGACY_LIBDIR}/cmake/odbcpp" NO_DEFAULT_PATH)
_legacy_prefix_identity("${odbcpp_CONFIG}" _selected_config)
_legacy_prefix_identity("${_package}/odbcppConfig.cmake" _expected_config)
if(NOT _selected_config STREQUAL _expected_config)
  message(FATAL_ERROR "Actual selected package configuration escapes prefix")
endif()
get_property(_imported DIRECTORY PROPERTY IMPORTED_TARGETS)
foreach(_target IN LISTS _imported)
  if(_target MATCHES "^odbcpp::" AND NOT _target MATCHES "^odbcpp::odbcpp_(core|driver)$")
    message(FATAL_ERROR "Extra exported SDK/auth/private target")
  endif()
endforeach()
set(_properties "")
foreach(_target IN ITEMS odbcpp::odbcpp_core odbcpp::odbcpp_driver)
  if(NOT TARGET ${_target})
    message(FATAL_ERROR "Missing legacy imported target: ${_target}")
  endif()
  foreach(_name IN ITEMS INTERFACE_INCLUDE_DIRECTORIES INTERFACE_LINK_LIBRARIES
      INTERFACE_COMPILE_DEFINITIONS INTERFACE_COMPILE_OPTIONS INTERFACE_LINK_OPTIONS IMPORTED_CONFIGURATIONS IMPORTED_LOCATION
      IMPORTED_LOCATION_RELEASE IMPORTED_LOCATION_DEBUG IMPORTED_IMPLIB_RELEASE)
    get_target_property(_value ${_target} ${_name})
    string(APPEND _properties "${_target}.${_name}=${_value}\n")
    foreach(_forbidden IN LISTS LEGACY_FORBIDDEN_ROOTS)
      if(_forbidden)
        string(FIND "${_value}" "${_forbidden}" _found)
        if(NOT _found EQUAL -1)
          message(FATAL_ERROR "Imported target references forbidden source/build/original prefix")
        endif()
      endif()
    endforeach()
  endforeach()
endforeach()
string(APPEND _properties "compiler=${CMAKE_CXX_COMPILER}\nimplicit_includes=${CMAKE_CXX_IMPLICIT_INCLUDE_DIRECTORIES}\nimplicit_link_dirs=${CMAKE_CXX_IMPLICIT_LINK_DIRECTORIES}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/compiler-identity.txt" "${CMAKE_CXX_COMPILER}\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/imported-properties.txt" "selected_config=${_selected_config}\n${_properties}")
if(LEGACY_MODE STREQUAL "CURRENT_ROOT_NEGATIVE" OR LEGACY_MODE STREQUAL "CORRECTED_ROOT_POSITIVE")
  set(_header contract_first_include.cpp)
elseif(LEGACY_MODE STREQUAL "MYSQL_CRYPTO_NEGATIVE")
  set(_header mysql_first_include.cpp)
else()
  message(FATAL_ERROR "Explicit closed consumer mode required")
endif()
add_library(legacy_header OBJECT "${_header}")
target_compile_features(legacy_header PRIVATE cxx_std_20)
target_compile_options(legacy_header PRIVATE -MD)
target_link_libraries(legacy_header PRIVATE odbcpp::odbcpp_core)
add_executable(legacy_driver_link_only EXCLUDE_FROM_ALL driver_link_only.cpp)
target_compile_features(legacy_driver_link_only PRIVATE cxx_std_20)
target_compile_options(legacy_driver_link_only PRIVATE -MD)
target_link_libraries(legacy_driver_link_only PRIVATE odbcpp::odbcpp_driver)
foreach(_target IN ITEMS legacy_header legacy_driver_link_only)
  if(MSVC)
    target_compile_options(${_target} PRIVATE /W4 /WX /permissive- /utf-8 /wd4100 /wd4189)
  else()
    target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic -Werror
      -Wno-unused-parameter -Wno-unused-variable -Wno-unused-function)
  endif()
endforeach()
# No add_test, execution, include injection, new exported target or SDK linkage.
