cmake_minimum_required(VERSION 3.20)
project(odbcpp_installed_setup_tests LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Installed MSI setup qualification requires native Windows")
endif()
set(TARGET_DATABASE "" CACHE STRING "Installed package product")
set_property(CACHE TARGET_DATABASE PROPERTY STRINGS POSTGRESQL REDSHIFT)
if(NOT TARGET_DATABASE STREQUAL "POSTGRESQL" AND NOT TARGET_DATABASE STREQUAL "REDSHIFT")
  message(FATAL_ERROR "Specify TARGET_DATABASE=POSTGRESQL or REDSHIFT")
endif()
option(ODBCPP_INSTALLED_SETUP_TESTS "Run against the matching installed MSI" ON)
if(NOT ODBCPP_INSTALLED_SETUP_TESTS)
  message(FATAL_ERROR "This entry point exists only for explicit installed-MSI qualification")
endif()
include(FetchContent)
FetchContent_Declare(googletest
  URL https://github.com/google/googletest/archive/refs/tags/v1.14.0.zip
  DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
enable_testing()
add_executable(it_setup_registration ../it_setup_registration.cpp)
target_compile_features(it_setup_registration PRIVATE cxx_std_20)
target_link_libraries(it_setup_registration PRIVATE odbccp32 advapi32 GTest::gtest_main)
if(TARGET_DATABASE STREQUAL "REDSHIFT")
  target_compile_definitions(it_setup_registration PRIVATE ODBCPP_TEST_REDSHIFT_PACKAGE=1)
endif()
if(MSVC)
  target_compile_options(it_setup_registration PRIVATE /W4 /WX /utf-8)
endif()
add_test(NAME it_setup_registration COMMAND it_setup_registration)
set_tests_properties(it_setup_registration PROPERTIES LABELS "integration;windows;package" TIMEOUT 30)
